## 安装 huggingface-cli

```bash
# 配置虚拟环境
python -m venv ~/ollama-env
source ollama-env/bin/activate

# 安装 hf cli
pip install -U "huggingface_hub[cli]"

# 配置镜像并下载
export HF_ENDPOINT=https://hf-mirror.com
hf download Qwen/Qwen3-VL-4B-Instruct

# 下载好的文件存放在 ～/.cache/huggingface/hub/里面，可以通过下面的命令查看
ls -lh /root/.cache/huggingface/hub/models--GatekeeperZA--Qwen3-VL-4B-Instruct-RKLLM-v1.2.3/snapshots/9bf1a42f67a15261a88dbae4e0cfae3ae41ae843/

# 下载模型转化工具
git clone git@github.com:alibaba/MNN.git

# 安装依赖
cd ~/MNN/transformers/llm/export
pip install -r requirements.txt

# 转换模型
python3 MNN/transformers/llm/export/llmexport.py \
    --path /root/.cache/huggingface/hub/models--Qwen--Qwen3-VL-4B-Instruct/snapshots/ebb281ec70b05090aa6165b016eac8ec08e71b17 \
    --export mnn \
    --quant_bit 4 \
    --quant_block 128 \
    --dst_path ./Qwen3-VL-4B-MNN

# 编译 MNN 执行器
cd ~/MNN
mkdir build && cd build
# 配置 CMake，关键选项说明：
# -DMNN_BUILD_LLM=ON : 开启大语言模型支持
# -DMNN_BUILD_LLM_OMNI=ON : 开启多模态(视觉/音频)支持
# -DMNN_LOW_MEMORY=ON : 低内存模式，对你 16GB 的板子更友好
# -DLLM_SUPPORT_VISION=ON : 显式开启视觉支持
cmake .. -DMNN_BUILD_LLM=ON -DMNN_BUILD_LLM_OMNI=ON -DMNN_LOW_MEMORY=ON -DLLM_SUPPORT_VISION=ON
# 开始编译，-j4 表示用 4 个核心并行编译以加快速度
make -j4 llm_demo

#测试是否成功
~/MNN/build/llm_demo ～/Qwen3-VL-4B-MNN/config.json 
```
